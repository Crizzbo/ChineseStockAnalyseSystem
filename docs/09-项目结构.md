#### 3.2.3 服务层示例
```python
# app/services/stock_service.py
from typing import List, Optional
from app.repositories.stock_repository import StockRepository
from app.external.stock_data_providers import StockDataProvider
from app.schemas.stock import StockInfo, StockPrice
from app.utils.cache_utils import CacheManager

class StockService:
    def __init__(
        self,
        stock_repo: StockRepository,
        data_provider: StockDataProvider,
        cache_manager: CacheManager
    ):
        self.stock_repo = stock_repo
        self.data_provider = data_provider
        self.cache = cache_manager

    async def get_stock_info(self, symbol: str) -> Optional[StockInfo]:
        """获取股票基本信息"""
        # 先从缓存获取
        cached_info = await self.cache.get_stock_info(symbol)
        if cached_info:
            return cached_info

        # 从数据库获取
        stock_info = await self.stock_repo.get_by_symbol(symbol)
        if stock_info:
            await self.cache.set_stock_info(symbol, stock_info)
            return stock_info

        # 从外部API获取并保存
        external_info = await self.data_provider.get_stock_info(symbol)
        if external_info:
            saved_info = await self.stock_repo.create(external_info)
            await self.cache.set_stock_info(symbol, saved_info)
            return saved_info

        return None

    async def search_stocks(self, keyword: str, limit: int = 20) -> List[StockInfo]:
        """搜索股票"""
        return await self.stock_repo.search(keyword, limit)

    async def get_real_time_price(self, symbol: str) -> Optional[StockPrice]:
        """获取实时价格"""
        # 从缓存获取实时价格
        cached_price = await self.cache.get_stock_price(symbol)
        if cached_price:
            return cached_price

        # 从外部API获取
        price_data = await self.data_provider.get_real_time_price(symbol)
        if price_data:
            await self.cache.set_stock_price(symbol, price_data, ttl=30)
            return price_data

        return None
```

### 6.2 脚本示例
```python
# scripts/data_collection/sync_stock_list.py
#!/usr/bin/env python3
"""
同步股票列表脚本
"""

import asyncio
import logging
from app.external.stock_data_providers import StockDataProvider
from app.repositories.stock_repository import StockRepository
from app.core.database import get_database

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)

async def sync_stock_list():
    """同步股票列表"""
    db = get_database()
    stock_repo = StockRepository(db)
    data_provider = StockDataProvider()

    try:
        # 获取股票列表
        logger.info("Fetching stock list from external API...")
        stock_list = await data_provider.get_all_stocks()

        # 批量更新数据库
        logger.info(f"Updating {len(stock_list)} stocks in database...")
        for stock in stock_list:
            await stock_repo.upsert(stock)

        logger.info("Stock list sync completed successfully")

    except Exception as e:
        logger.error(f"Error syncing stock list: {e}")
        raise
    finally:
        await db.close()

if __name__ == "__main__":
    asyncio.run(sync_stock_list())
```
