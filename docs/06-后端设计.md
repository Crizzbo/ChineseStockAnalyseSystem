- **核心框架**: Python + Flask/FastAPI

### 4.3 AkShare数据质量评估
```python
class AkShareDataQualityService:
    def __init__(self):
        self.analysis_system = StockAnalysisSystem()
        self.validator = DataValidator()

    def assess_data_quality(self, symbol: str) -> dict:
        """评估AkShare数据质量"""
        quality_report = {
            'symbol': symbol,
            'data_completeness': {},
            'data_freshness': {},
            'data_consistency': {},
            'quality_score': 0.0
        }

        try:
            # 检查实时数据
            realtime_data = self.analysis_system.realtime_service.get_stock_realtime(symbol)
            quality_report['data_completeness']['realtime'] = realtime_data is not None

            # 检查历史数据
            historical_data = self.analysis_system.historical_service.get_recent_data(symbol, 30)
            quality_report['data_completeness']['historical'] = historical_data is not None and not historical_data.empty

            # 检查财务数据
            financial_data = self.analysis_system.fundamental_service.get_comprehensive_financial_data(symbol)
            quality_report['data_completeness']['financial'] = financial_data is not None

            # 数据验证
            if historical_data is not None:
                validation_result = self.validator.validate_stock_data(historical_data, 'historical')
                quality_report['data_consistency'] = validation_result

            # 计算质量评分
            quality_report['quality_score'] = self._calculate_quality_score(quality_report)

            return quality_report

        except Exception as e:
            quality_report['error'] = str(e)
            return quality_report

    def _calculate_quality_score(self, quality_report: dict) -> float:
        """计算数据质量评分"""
        completeness = quality_report['data_completeness']
        consistency = quality_report.get('data_consistency', {})

        # 完整性评分 (40%)
        completeness_score = sum([1 for v in completeness.values() if v]) / len(completeness)

        # 一致性评分 (60%)
        consistency_score = 1.0 if consistency.get('is_valid', False) else 0.5

        return completeness_score * 0.4 + consistency_score * 0.6
```

def calculate_cost_basis(transactions):
    """计算持仓成本"""
    total_shares = 0
    total_cost = 0

    for transaction in transactions:
        if transaction.type == 'BUY':
            shares = transaction.quantity
            cost = transaction.price * shares + transaction.fees
            total_cost += cost
            total_shares += shares
        elif transaction.type == 'SELL':
            # 处理卖出逻辑 (先进先出)
            pass

    return total_cost / total_shares if total_shares > 0 else 0

def calculate_portfolio_performance(holdings, current_prices):
    """计算组合收益 - 使用AkShare实时价格"""
    total_cost = 0
    total_value = 0

    for holding in holdings:
        cost = holding.cost_basis * holding.quantity
        # 使用AkShare实时价格
        current_price = current_prices.get(holding.symbol, holding.cost_basis)
        value = current_price * holding.quantity

        total_cost += cost
        total_value += value

    return {
        'total_cost': total_cost,
        'total_value': total_value,
        'total_return': total_value - total_cost,
        'return_rate': (total_value - total_cost) / total_cost if total_cost > 0 else 0,
        'data_source': 'akshare_realtime'
    }

class AkShareDataCleaner:
    def clean_financial_data(self, raw_data):
        """清洗AkShare财务数据"""
        cleaned_data = {}

        # AkShare特定的数据类型转换
        for key, value in raw_data.items():
            if isinstance(value, str):
                if value.isdigit() or self._is_float(value):
                    cleaned_data[key] = float(value)
                elif value in ['--', 'N/A', '', '暂无数据']:
                    cleaned_data[key] = None
                else:
                    cleaned_data[key] = value
            else:
                cleaned_data[key] = value

        # 数据验证
        return self._validate_akshare_data(cleaned_data)

    def _validate_akshare_data(self, data):
        """AkShare数据验证"""
        # 检查必要字段
        required_fields = ['营业收入', '净利润', '总资产']
        for field in required_fields:
            if field not in data or data[field] is None:
                logger.warning(f"AkShare数据缺少必要字段: {field}")

        # 逻辑检查
        if data.get('净利润', 0) > data.get('营业收入', 0):
            logger.warning("净利润超过营业收入，请检查AkShare数据")

        return data

        logger.info(f"AkShare实时数据同步完成: 成功{success_count}个, 失败{error_count}个")
        return {'success': success_count, 'errors': error_count}

    except Exception as e:
        logger.error(f"AkShare实时数据同步任务失败: {e}")
        raise
