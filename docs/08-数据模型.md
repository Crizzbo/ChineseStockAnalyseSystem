# 数据模型设计

## 1. 数据库设计概览

### 1.1 数据库选择
- **主数据库**：PostgreSQL（支持复杂查询和事务）
- **缓存数据库**：Redis（高性能缓存和会话存储）
- **时序数据库**：InfluxDB（时间序列数据存储）
- **搜索引擎**：Elasticsearch（全文搜索和分析）

### 1.2 数据库架构
```
┌─────────────────────────────────────────────────────────────┐
│                        应用层                                 │
├─────────────────────────────────────────────────────────────┤
│  PostgreSQL        │  Redis         │  InfluxDB            │
│  (主要业务数据)     │  (缓存/会话)    │  (时间序列数据)       │
│  - 用户数据         │  - 实时价格     │  - 股价历史          │
│  - 股票基础信息     │  - 分析结果     │  - 技术指标          │
│  - 财务报表         │  - 用户会话     │  - 交易量            │
│  - 分析报告         │                │                     │
└─────────────────────────────────────────────────────────────┘
```

## 2. PostgreSQL数据模型

### 2.1 用户相关表

#### 2.1.1 用户表 (users)
```sql
CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    username VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(100) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(100),
    phone VARCHAR(20),
    avatar_url VARCHAR(255),
    status VARCHAR(20) DEFAULT 'active',
    risk_preference VARCHAR(20) DEFAULT 'moderate', -- conservative, moderate, aggressive
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    last_login_at TIMESTAMP
);

-- 索引
CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_username ON users(username);
CREATE INDEX idx_users_status ON users(status);
```

```sql
CREATE TABLE user_preferences (
```

## 6. 数据操作接口

### 6.1 数据访问层 (Repository Pattern)

#### 6.1.1 股票数据仓库
```python
class StockRepository:
    def __init__(self, db_session):
        self.db = db_session

    async def get_stock_info(self, symbol: str) -> Optional[Stock]:
        """获取股票基本信息"""
        return await self.db.query(Stock).filter(
            Stock.symbol == symbol
        ).first()

    async def search_stocks(self, keyword: str, limit: int = 20) -> List[Stock]:
        """搜索股票"""
        return await self.db.query(Stock).filter(
            or_(
                Stock.symbol.ilike(f"%{keyword}%"),
                Stock.name.ilike(f"%{keyword}%")
            )
        ).limit(limit).all()

    async def get_stocks_by_industry(self, industry: str) -> List[Stock]:
        """按行业获取股票"""
        return await self.db.query(Stock).filter(
            Stock.industry == industry
        ).all()
```

#### 6.1.2 财务数据仓库
```python
class FinancialRepository:
    def __init__(self, db_session):
        self.db = db_session

    async def get_latest_financial_data(self, symbol: str) -> Optional[FinancialStatement]:
        """获取最新财务数据"""
        return await self.db.query(FinancialStatement).filter(
            FinancialStatement.symbol == symbol
        ).order_by(
            FinancialStatement.report_year.desc(),
            FinancialStatement.report_period.desc()
        ).first()

    async def get_financial_history(self, symbol: str, years: int = 5) -> List[FinancialStatement]:
        """获取历史财务数据"""
        cutoff_year = datetime.now().year - years
        return await self.db.query(FinancialStatement).filter(
            FinancialStatement.symbol == symbol,
            FinancialStatement.report_year >= cutoff_year
        ).order_by(
            FinancialStatement.report_year.desc(),
            FinancialStatement.report_period.desc()
        ).all()
```

### 6.2 缓存操作接口

#### 6.2.1 Redis缓存管理
```python
class CacheManager:
    def __init__(self, redis_client):
        self.redis = redis_client

    async def get_stock_price(self, symbol: str) -> Optional[dict]:
        """获取股票实时价格缓存"""
        key = f"stock:realtime:{symbol}"
        data = await self.redis.hgetall(key)
        return data if data else None

    async def set_stock_price(self, symbol: str, price_data: dict, ttl: int = 30):
        """设置股票实时价格缓存"""
        key = f"stock:realtime:{symbol}"
        await self.redis.hset(key, mapping=price_data)
        await self.redis.expire(key, ttl)

    async def get_analysis_result(self, symbol: str, analysis_type: str) -> Optional[dict]:
        """获取分析结果缓存"""
        key = f"analysis:{symbol}:{analysis_type}"
        data = await self.redis.get(key)
        return json.loads(data) if data else None

    async def set_analysis_result(self, symbol: str, analysis_type: str,
                                  result: dict, ttl: int = 3600):
        """设置分析结果缓存"""
        key = f"analysis:{symbol}:{analysis_type}"
        await self.redis.setex(key, ttl, json.dumps(result, ensure_ascii=False))
```

### 6.3 时序数据操作

#### 6.3.1 InfluxDB操作
```python
class TimeSeriesRepository:
    def __init__(self, influx_client):
        self.client = influx_client

    async def write_stock_price(self, symbol: str, price_data: dict):
        """写入股票价格数据"""
        point = Point("stock_prices") \
            .tag("symbol", symbol) \
            .tag("exchange", price_data.get("exchange")) \
            .field("open", float(price_data["open"])) \
            .field("high", float(price_data["high"])) \
            .field("low", float(price_data["low"])) \
            .field("close", float(price_data["close"])) \
            .field("volume", int(price_data["volume"])) \
            .time(datetime.utcnow())

        await self.client.write_api().write(point)

    async def get_price_history(self, symbol: str, start_time: datetime,
                                end_time: datetime) -> List[dict]:
        """获取股票价格历史数据"""
        query = f'''
        from(bucket: "stock_data")
        |> range(start: {start_time.isoformat()}, stop: {end_time.isoformat()})
        |> filter(fn: (r) => r._measurement == "stock_prices")
        |> filter(fn: (r) => r.symbol == "{symbol}")
        |> pivot(rowKey:["_time"], columnKey: ["_field"], valueColumn: "_value")
        '''

        result = await self.client.query_api().query(query)
        return [record.values for record in result[0].records]
```
